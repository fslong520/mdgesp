   4    struct Human {

  10    void bubble_sort(int n) {
  11        for (int i = 0; i < n; ++i) {
  12            for (int j = 1; j < n - i; ++j) {
  13                double bmi1 = (double)humans[j-1].w / humans[j-1].h / humans[j-1].h;
