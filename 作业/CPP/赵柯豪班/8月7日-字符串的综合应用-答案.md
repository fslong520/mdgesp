```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    string s, ans = "";
    getline(cin, s);
    for (char c : s) {
        if (c >= '0' && c <= '9') {
            ans += c;
        } else if (ans.empty() || ans[ans.size() - 1] != '*') {
            ans += '*';
        }
    }
    cout << ans << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin >> n;
    string names[105];
    for (int i = 0; i < n; ++i) cin >> names[i];
    for (int i = 0; i < n; ++i) {
        int k = i;
        for (int j = i + 1; j < n; ++j)
            if (names[j] < names[k]) k = j;
        swap(names[i], names[k]);
    }
    for (int i = 0; i < n; ++i) cout << names[i] << endl;
    return 0;
}
```

#include <bits/stdc++.h>
using namespace std;
int main() {
    string s, a, b;
    getline(cin, s);
    getline(cin, a);
    getline(cin, b);
    size_t pos = 0;
    while ((pos = s.find(a, pos)) != string::npos) {
        s.replace(pos, a.length(), b);
        pos += b.length();
    }
    cout << s << endl;
    return 0;
}
