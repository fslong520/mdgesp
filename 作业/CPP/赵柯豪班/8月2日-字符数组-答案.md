18. cpp (100分)

参考代码:
```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    char s[105];
    cin >> s;
    int low = 0, up = 0;
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] >= 'a' && s[i] <= 'z') low++;
        else if (s[i] >= 'A' && s[i] <= 'Z') up++;
    }
    cout << low << " " << up << endl;
    return 0;
}
```
