#include <bits/stdc++.h>

    int n;
    string s;
    cin >> n;
    cin.ignore();
    getline(cin, s);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < i; ++j) cout << ' ';
        cout << s << endl;
    }
    return 0;
}
