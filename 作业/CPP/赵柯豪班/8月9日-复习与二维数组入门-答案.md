    int n, m, k;
    cin >> n >> m;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            cin >> a[i][j];
    cin >> k;
    int cnt = 0;
    for (int i = 0; i < n; i++)

}

    bool first = true;

            if (!first) cout << ' ';

        }
