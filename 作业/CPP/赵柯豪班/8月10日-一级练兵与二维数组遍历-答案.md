    int cnt = 0;
    int now = w;
    if (now >= 3600 && now <= 3700) cnt++;
    for (int i = 2; i <= n; i++) {
        int x;

20. cpp (100分)

参考代码:
```cpp
#include <bits/stdc++.h>
using namespace std;
int a[105][105];
int main() {
    int n, m;
    cin >> n >> m;
    int mx = -1, mi = 0, mj = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];
            if (a[i][j] > mx) {
                mx = a[i][j];
                mi = i;
                mj = j;
            }
        }
    }
    cout << mx << endl;
    cout << mi + 1 << " " << mj + 1 << endl;
    return 0;
}
```
