### 程序一：Kahn 算法求拓扑排序（10 分）

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100005;
vector<int> g[N];
int in[N];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        in[v]++;
    }
    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (in[i] == 0) q.push(i);
    int cnt = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        cout << u << " ";
        cnt++;
        for (int v : g[u]) {
            in[v]--;
            if (in[v] == 0) q.push(v);
        }
    }
    cout << endl;
    if (cnt < n) cout << "Cycle" << endl;
    return 0;
}
```

### 程序二：DFS 判环并求拓扑序（15 分）

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100005;
vector<int> g[N];
int vis[N];          // 0 未访问，1 访问中，2 已结束
vector<int> ans;

bool dfs(int u) {
    vis[u] = 1;
    for (int v : g[u]) {
        if (vis[v] == 1) return false;   // 有环
        if (vis[v] == 0) {
            if (!dfs(v)) return false;
        }
    }
    vis[u] = 2;
    ans.push_back(u);
    return true;
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
    }
    bool ok = true;
    for (int i = 1; i <= n; i++)
        if (vis[i] == 0) {
            if (!dfs(i)) { ok = false; break; }
        }
    if (!ok) cout << "Cycle" << endl;
    else {
        reverse(ans.begin(), ans.end());
        for (int x : ans) cout << x << " ";
        cout << endl;
    }
    return 0;
}
```

## 拓扑排序知识点总结

### 1. 拓扑排序定义

- **拓扑排序**：对有向无环图（DAG）的所有顶点排成一个线性序列，使得图中每条边 u→v，顶点 u 都排在 v 之前
- **适用条件**：仅 DAG 可拓扑排序；有向图存在拓扑排序 ⟺ 无有向环
- 拓扑排序结果**不一定唯一**：当多个入度为 0 的顶点同时存在时，顺序可不同
- 若有 n 个顶点的图能排出 n 个顶点的拓扑序，则必无环

### 2. Kahn 算法（BFS 思路）

- 统计每个顶点的**入度**（指向它的边数）
- 将入度为 0 的顶点入队（它们没有前置依赖，可先处理）
- 循环：出队一个顶点 u 并输出；对 u 的每个后继 v，将 in[v] 减 1，若 in[v] 变为 0 则入队
- 结束时若输出的顶点数 < n，则图中有环
- **时间复杂度**：O(n + m)（邻接表）

### 3. DFS 求拓扑序（判环）

- 三色标记：0 未访问、1 访问中（在递归栈内）、2 已结束
- 若在 DFS 中遇到 vis[v] == 1，说明回到递归栈内顶点，存在有向环
- 结束访问时把顶点加入序列（后序），最后**逆序**输出即为拓扑序
- 复杂度同为 O(n + m)

### 4. 字典序最小的拓扑序

- Kahn 算法的普通队列改为**小顶堆**（priority_queue，greater）
- 每次取编号最小的入度为 0 顶点输出，可得到字典序最小的拓扑序

### 5. 拓扑排序的应用

| 应用 | 说明 |
|------|------|
| 选课/先修安排 | 判断能否修完所有课程，输出学习顺序 |
| 任务调度 | 依赖关系下的执行顺序 |
| 判环 | Kahn 输出数 < n 或 DFS 遇访问中顶点 |
| 编译/构建顺序 | 依赖库先编译 |

#### Kahn 算法（含判环）

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100005;
vector<int> g[N];
int in[N];

int main() {
```

#### 字典序最小拓扑序

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100005;
vector<int> g[N];
int in[N];

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        in[v]++;
    }
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int i = 1; i <= n; i++)
        if (in[i] == 0) pq.push(i);
    vector<int> ans;
    while (!pq.empty()) {
        int u = pq.top(); pq.pop();
        ans.push_back(u);
        for (int v : g[u]) {
            in[v]--;
            if (in[v] == 0) pq.push(v);
        }
    }
    for (int x : ans) cout << x << " ";
    cout << "\n";
    return 0;
}
```

#### DFS 判环 + 拓扑序

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 100005;
vector<int> g[N];
int vis[N];
vector<int> ans;

bool dfs(int u) {
    vis[u] = 1;
    for (int v : g[u]) {
        if (vis[v] == 1) return false;
        if (vis[v] == 0)
            if (!dfs(v)) return false;
    }
    vis[u] = 2;
    ans.push_back(u);
    return true;
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
    }
    bool ok = true;
    for (int i = 1; i <= n; i++)
        if (vis[i] == 0)
            if (!dfs(i)) { ok = false; break; }
    if (!ok) {
        cout << "Cycle\n";
    } else {
        reverse(ans.begin(), ans.end());
        for (int x : ans) cout << x << " ";
        cout << "\n";
    }
    return 0;
}
```

最后一行 n 个整数，表示待验证的排列。

**输出格式**

若是合法拓扑序列，输出 `Yes`；否则输出 `No`。

**样例输入 1**

```
5 4
1 2
1 3
2 4
3 4
1 2 3 4 5
```

**样例输出 1**

```
Yes
```

**样例解释 1**

1 在 2、3 前，2、3 在 4 前，5 无依赖。排列 1 2 3 4 5 满足所有先后关系。

**样例输入 2**

```
5 4
1 2
1 3
2 4
3 4
1 2 4 3 5
```

**样例输出 2**

```
No
```

**样例解释 2**

先后关系 3→4 要求 3 排在 4 前，但排列中 4 在 3 前，故不合法。

**提示**

- 先读入排列，用 pos 数组记录每个数字在排列中的位置
- 再对每条边 u→v，检查 pos[u] 是否小于 pos[v]
- 若存在一条边 u→v 满足 pos[u] > pos[v]，则排列不合法
- 复杂度 O(n + m)

**数据范围**

对于 30% 的数据，1 ≤ n ≤ 100，1 ≤ m ≤ 500。  
对于 60% 的数据，1 ≤ n ≤ 10⁴，1 ≤ m ≤ 5 × 10⁴。  
对于 100% 的数据，1 ≤ n ≤ 10⁵，1 ≤ m ≤ 10⁵。
